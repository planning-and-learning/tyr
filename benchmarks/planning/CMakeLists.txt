add_custom_target(benchmarks_planning)

function(tyr_add_benchmark name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE tyr::core benchmark::benchmark Boost::json)
    target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/benchmarks/support)
    add_dependencies(benchmarks_planning ${name})
endfunction()

foreach(backend ground lifted)
    if(backend STREQUAL "ground")
        set(task_kind GroundTag)
    else()
        set(task_kind LiftedTag)
    endif()

    tyr_add_benchmark(astar_blind_${backend} algorithms/astar_eager/astar_blind.cpp)
    target_compile_definitions(astar_blind_${backend} PRIVATE TYR_BENCHMARK_TASK_KIND=${task_kind})

    tyr_add_benchmark(successors_${backend} successor_generator.cpp)
    tyr_add_benchmark(schema_selection_${backend} ${backend}/schema_selection.cpp)

    foreach(heuristic rpg_add rpg_ff lmcut)
        if(heuristic STREQUAL "rpg_add")
            set(heuristic_type AddRPGHeuristic)
        elseif(heuristic STREQUAL "rpg_ff")
            set(heuristic_type FFRPGHeuristic)
        else()
            set(heuristic_type LMCutHeuristic)
        endif()
        tyr_add_benchmark(gbfs_${heuristic}_${backend} algorithms/gbfs_lazy/gbfs.cpp)
        target_compile_definitions(gbfs_${heuristic}_${backend} PRIVATE TYR_BENCHMARK_HEURISTIC=${heuristic_type})
        if(backend STREQUAL "ground")
            target_compile_definitions(gbfs_${heuristic}_${backend} PRIVATE TYR_BENCHMARK_GROUND)
        endif()
    endforeach()
    if(backend STREQUAL "ground")
        target_compile_definitions(successors_${backend} PRIVATE TYR_BENCHMARK_GROUND)
    endif()
endforeach()
